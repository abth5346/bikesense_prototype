// ============================================================ // BLUE GECKO BIKE - BLE // Cycling Speed & Cadence prototype // ============================================================ // ----------------------------- // BLE UUIDs // ----------------------------- const CSC_SERVICE_UUID = '00001816-0000-1000-8000-00805f9b34fb'; const CSC_MEASUREMENT_UUID = '00002a5b-0000-1000-8000-00805f9b34fb'; // ----------------------------- // Bike configuration // ----------------------------- // Wheel circumference in meters. // 2.11 m ≈ 700c road-bike wheel. const WHEEL_CIRCUMFERENCE_M = 2.11; // ----------------------------- // BLE state // ----------------------------- let device = null; let server = null; let measurementCharacteristic = null; // ----------------------------- // Measurement state // ----------------------------- let previousWheelRevolutions = null; let previousWheelEventTime = null; let previousCrankRevolutions = null; let previousCrankEventTime = null; // ----------------------------- // DOM elements // ----------------------------- const statusPanel = document.getElementById('statusPanel'); const actionButton = document.getElementById('actionButton'); const speedValue = document.getElementById('speedVal'); const cadenceValue = document.getElementById('cadenceVal'); const revolutionsValue = document.getElementById('revsVal'); const linkValue = document.getElementById('linkVal'); const consoleOutput = document.getElementById('console'); // ============================================================ // LOGGING // ============================================================ function log(message) { const timestamp = new Date().toLocaleTimeString(); consoleOutput.textContent += `\n[${timestamp}] ${message}`; consoleOutput.scrollTop = consoleOutput.scrollHeight; console.log(`[${timestamp}] ${message}`); } // ============================================================ // BUTTON // ============================================================ actionButton.addEventListener( 'click', async () => { if (device?.gatt?.connected) { disconnectBike(); } else { await connectBike(); } } ); // ============================================================ // CONNECT // ============================================================ async function connectBike() { if (!navigator.bluetooth) { log( 'ERROR: Web Bluetooth is not supported.' ); statusPanel.textContent = 'Error: Web Bluetooth Not Supported'; return; } try { log('Opening Bluetooth device picker...'); statusPanel.textContent = 'Select your bike...'; // ---------------------------------------------------- // For LightBlue testing: // acceptAllDevices allows us to see the peripheral // without filtering by its advertised service. // ---------------------------------------------------- device = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: [ CSC_SERVICE_UUID ] }); log( `Selected device: ${ device.name || 'Unknown Device' }` ); device.addEventListener( 'gattserverdisconnected', handleDisconnect ); await connectGatt(); await discoverGatt(); await startNotifications(); setConnectedState(); } catch (error) { log( `Connection failed: ${error.message}` ); resetBleState(); } } // ============================================================ // GATT CONNECTION // ============================================================ async function connectGatt() { log('Connecting to GATT server...'); server = await device.gatt.connect(); log('GATT connection established.'); } // ============================================================ // GATT DISCOVERY // ============================================================ async function discoverGatt() { log( 'Searching for Cycling Speed & Cadence service...' ); const service = await server.getPrimaryService( CSC_SERVICE_UUID ); log( 'CSC service found.' ); log( 'Searching for CSC Measurement characteristic...' ); measurementCharacteristic = await service.getCharacteristic( CSC_MEASUREMENT_UUID ); log( 'CSC Measurement characteristic found.' ); } // ============================================================ // NOTIFICATIONS // ============================================================ async function startNotifications() { log( 'Enabling measurement notifications...' ); await measurementCharacteristic .startNotifications(); measurementCharacteristic.addEventListener( 'characteristicvaluechanged', handleMeasurement ); log( 'Measurement notifications enabled.' ); } // ============================================================ // CSC MEASUREMENT PARSER // ============================================================ function handleMeasurement(event) { try { const data = event.target.value; const flags = data.getUint8(0); let offset = 1; // ---------------------------------------------------- // Wheel Revolution Data // ---------------------------------------------------- if (flags & 0x01) { const wheelRevolutions = data.getUint32( offset, true ); offset += 4; const wheelEventTime = data.getUint16( offset, true ); offset += 2; revolutionsValue.textContent = wheelRevolutions; calculateSpeed( wheelRevolutions, wheelEventTime ); } // ---------------------------------------------------- // Crank Revolution Data // ---------------------------------------------------- if (flags & 0x02) { const crankRevolutions = data.getUint16( offset, true ); offset += 2; const crankEventTime = data.getUint16( offset, true ); offset += 2; calculateCadence( crankRevolutions, crankEventTime ); } } catch (error) { log( `Packet parsing error: ${error.message}` ); } } // ============================================================ // SPEED CALCULATION // ============================================================ function calculateSpeed( currentRevolutions, currentEventTime ) { if ( previousWheelRevolutions === null || previousWheelEventTime === null ) { previousWheelRevolutions = currentRevolutions; previousWheelEventTime = currentEventTime; return; } const revolutionDelta = calculateCounterDelta( currentRevolutions, previousWheelRevolutions, 0x100000000 ); const timeTicks = calculateCounterDelta( currentEventTime, previousWheelEventTime, 0x10000 ); const timeSeconds = timeTicks / 1024; if (timeSeconds > 0) { const metersPerSecond = ( revolutionDelta * WHEEL_CIRCUMFERENCE_M ) / timeSeconds; const milesPerHour = metersPerSecond * 2.23694; speedValue.textContent = Math.max( 0, milesPerHour ).toFixed(1); } previousWheelRevolutions = currentRevolutions; previousWheelEventTime = currentEventTime; } // ============================================================ // CADENCE CALCULATION // ============================================================ function calculateCadence( currentRevolutions, currentEventTime ) { if ( previousCrankRevolutions === null || previousCrankEventTime === null ) { previousCrankRevolutions = currentRevolutions; previousCrankEventTime = currentEventTime; return; } const revolutionDelta = calculateCounterDelta( currentRevolutions, previousCrankRevolutions, 0x10000 ); const timeTicks = calculateCounterDelta( currentEventTime, previousCrankEventTime, 0x10000 ); const timeSeconds = timeTicks / 1024; if (timeSeconds > 0) { const rpm = ( revolutionDelta / timeSeconds ) * 60; cadenceValue.textContent = Math.round( Math.max(0, rpm) ); } previousCrankRevolutions = currentRevolutions; previousCrankEventTime = currentEventTime; } // ============================================================ // COUNTER ROLLOVER // ============================================================ function calculateCounterDelta( current, previous, modulus ) { let delta = current - previous; if (delta < 0) { delta += modulus; } return delta; } // ============================================================ // CONNECTED UI // ============================================================ function setConnectedState() { statusPanel.textContent = `Connected: ${ device.name || 'Bike' }`; statusPanel.classList.add( 'active' ); actionButton.textContent = 'Disconnect From Bike'; actionButton.classList.add( 'disconnect' ); linkValue.textContent = 'ON'; linkValue.style.color = 'var(--success)'; log( 'BLE connection established successfully.' ); } // ============================================================ // DISCONNECT // ============================================================ function disconnectBike() { if ( device?.gatt?.connected ) { log( 'Disconnecting from bike...' ); device.gatt.disconnect(); } } // ============================================================ // DISCONNECTED EVENT // ============================================================ function handleDisconnect() { log( 'BLE device disconnected.' ); resetBleState(); } // ============================================================ // RESET // ============================================================ function resetBleState() { statusPanel.textContent = 'System Status: Disconnected'; statusPanel.classList.remove( 'active' ); actionButton.textContent = 'Connect to Bike'; actionButton.classList.remove( 'disconnect' ); linkValue.textContent = 'OFF'; linkValue.style.color = 'var(--danger)'; speedValue.textContent = '--'; cadenceValue.textContent = '--'; revolutionsValue.textContent = '--'; // Reset measurement history previousWheelRevolutions = null; previousWheelEventTime = null; previousCrankRevolutions = null; previousCrankEventTime = null; measurementCharacteristic = null; server = null; }
